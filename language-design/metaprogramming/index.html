Language Design

Metaprogramming, Macros & Reflection

Hygienic syntactic macro expansion, full compile-time execution (@comptime), static @if branches, and RTTI reflection (@typename, @typeinfo, rt::any).

Overview

Lora provides three complementary compile-time and reflection mechanisms without an unhygienic C preprocessor:

MechanismSyntaxEvaluation TimeCapabilities
Hygienic macrosmacro swap(a, b) = { ... }, @swap(x, y)AST expansionHygienic local bindings (no variable capture)
Compile-time VM@comptime(expr), @if (cond) { ... }Typecheck / evalRuns loops, arrays, structs, and pure functions at compile time
Reflection@typename(T), @typeinfo(T), rt::anyCompile-time & runtimeInspects struct fields, offsets, sizes, and dynamic values

Macros, @comptime & Reflection

impl named_shape for sq { fn area(q: *sq) f64 = q.s * q.s; fn name(q: *sq) str = "square"; }

fn area_of(s: shape) f64 = s.area();

export fn main() void = { let q = sq { s = 3.0 } let n: named_shape = &q; let s: shape = n; // shape's methods are named_shape's too assert(s.area() == 9.0 && area_of(&q) == 9.0 && n.name() == "square"); }

text 

```lora
type getter = interface { fn get() int; }
type setter = interface { fn set(v: int) void; }
type both = interface { getter; setter; }

type cell = struct { v: int }
impl getter for cell { fn get(c: *cell) int = c.v; }
impl setter for cell { fn set(c: *cell, v: int) void = { c.v = v; } }

fn reset(s: setter) void = s.set(0);

export fn main() void = {
	let c = cell { v = 5 }
	let b: both = &c;          // made of the impls of its parts
	reset(b);                  // and passed as the part it embeds second
	assert(b.get() == 0);
}

Default methods

A method can have a default body, used by impls that leave it out. In the body, self is a pointer to the implementing type, so other methods called on it are that type's own:

lora 
type shape = interface {
	fn area() f64;
	fn name() str = "a shape";
	fn describe() str = self.area() > 10.0 ? "big" : "small";
}

type sq = struct { s: f64 }
type circle = struct { r: f64 }

impl shape for sq {
	fn area(q: *sq) f64 = q.s * q.s;
}

impl shape for circle {
	fn area(c: *circle) f64 = 3.0 * c.r * c.r;
	fn name(c: *circle) str = "circle";            // replaces the default
}

export fn main() void = {
	let q = sq { s = 4.0 }
	let c = circle { r = 1.0 }
	assert(q.name() == "a shape" && q.describe() == "big");
	assert(c.name() == "circle" && c.describe() == "small");
}

Defaults cross module boundaries with the interface. Generic interfaces cannot have default bodies yet.

Fields

An interface can require fields, name: T;. A type implementing it must have a field of that name and type, anywhere in its layout; through an interface value the field can be read and changed:

lora 
type named = interface {
	name: str;
	count: int;
	fn greet() str = self.name;
}

type user = struct { id: int, name: str, count: int }
type robot = struct { model: u8, count: int, name: str }

impl named for user {}
impl named for robot {
	fn greet(r: *robot) str = "beep";
}

fn bump(n: named) void = { n.count += 1; }

export fn main() void = {
	let u = user { id = 1, name = "ada", count = 0 }
	let r = robot { model = 7, count = 10, name = "r2" }
	let all: [_]named = [&u, &r];
	bump(all[0]);
	bump(all[1]);
	assert(all[0].name == "ada" && u.count == 1 && r.count == 11);
	assert(all[1].greet() == "beep");
}
lora 
type named = interface { name: str; }
type t = struct { name: int }
impl named for t {} // error: named needs t to have a field `name` of type str
export fn main() void = void;

15. Macros

macro name(params) = body; declares a macro and @name(args) expands it. Expansion happens while type checking, so the result is ordinary, fully checked code. Each use of a parameter is the caller's argument expression, so arguments can be assigned to, see the caller's names, and are evaluated where and as often as they are used:

lora 
macro swap(a, b) = {
	let tmp = a;          // the body's own names never capture the caller's
	a = b;
	b = tmp;
}
macro unless(cond, body) = if (!(cond)) body;
macro guard(cond) = if (!(cond)) return 0;      // returns from the caller

fn checked(x: int) int = {
	@guard(x > 0);
	return x;
}

export fn main() void = {
	let tmp = 1, b = 2;
	@swap(tmp, b);
	assert(tmp == 2 && b == 1);
	let ran = false;
	@unless(tmp > 5, ran = true);
	assert(ran && checked(-1) == 0 && checked(3) == 3);
}
lora 
macro twice(x) = x + x;
fn bump(p: *int) int = { *p += 1; return *p; }

export fn main() void = {
	let n = 0;
	let r = @twice(bump(&n));   // bump runs twice: 1 + 2
	assert(n == 2 && r == 3);
}

export macro makes a macro available to importers, called @module::name(...), or @name(...) after use module::{name}.

What macros cannot do:

lora 
macro twice(x) = x + x;
export fn main() void = {
	@twice(1, 2); // error: Macro '@twice' takes 1 argument, got 2
}
lora 
macro r(x) = @r(x); // error: Macro '@r' is expanded more than 64 levels deep
export fn main() void = { @r(1); }
  • A macro takes a fixed number of arguments: no variadic or optional ones.
  • Expansion nests at most 64 levels.
  • Macros are not values; they cannot be passed or stored.
  • Names in the body other than parameters resolve where the macro is used.
  • comptime, clone, typename, typeinfo and undefined are built-in macros; a module's own macros of those names take precedence.

16. Compile-time execution

@comptime(expr) runs an expression while compiling, calling ordinary functions, and puts the resulting constant in its place. Use it anywhere a constant is needed:

lora 
use strings;

type range = struct { lo: int, hi: int }

fn fib(n: int) int = if (n < 2) n else fib(n - 1) + fib(n - 2);

fn squares() [5]int = {
	let out: [5]int = [0...];
	for (let i = 0z; i < 5; i += 1) out[i] = @int(i * i);
	return out;
}

fn bounds() range = range { lo = 1, hi = fib(10) }

def FIB20: int = @comptime(fib(20));                 // 6765, computed once

export fn main() void = {
	let table: [@comptime(fib(6))]u8 = [0...];        // an 8-element array

Valid Usage & Examples

Hygienic macros without variable capture

Local variables declared inside a macro body (such as tmp) never collide with variables of the same name at the call site.

lora Allowed 
macro square(x) = x * x;
macro swap(a, b) = {
	let tmp = a;
	a = b;
	b = tmp;
}

export fn main() void = {
	assert(@square(1 + 2) == 9);
	let x = 1, tmp = 2;
	@swap(x, tmp); // hygienic: macro's `tmp` does not collide with caller's `tmp`
	assert(x == 2 && tmp == 1);
}

Compile-time computation with @comptime and static @if

Compute lookup tables, array sizes, and def constants by running normal Lora functions inside @comptime(...), and prune dead branches with @if.

lora Allowed 
fn fib(n: int) int = if (n < 2) n else fib(n - 1) + fib(n - 2);
def FIB10: int = @comptime(fib(10));

export fn main() void = {
	let buf: [@comptime(fib(6))]u8 = [0...];
	assert(FIB10 == 55 && len(buf) == 8z);

	@if (size_of(usize) == 8) {
		assert(FIB10 == 55);
	} else {
		abort();
	}
}

Type introspection with @typename and @typeinfo

Query canonical type names, kinds, byte sizes, and struct field offsets at compile time or runtime.

lora Allowed 
use rt;

type Pair = struct { a: u8, b: u64 }

export fn main() void = {
	let info = @typeinfo(Pair);
	assert(info.kind == rt::type_kind::STRUCT && info.bytes == 16);
	assert(info.fields[1].name == "b" && info.fields[1].at == 8);
	assert(@typename([]Pair) == "[]Pair");
}

Disallowed Forms & Diagnostics

Exceeding @comptime step/depth limits or failing an assert at compile time

Infinite loops, unbounded recursion, or failed assert calls inside @comptime(...) fail compilation.

Compile error: Assertion failed / comptime execution limit exceeded
lora Rejected 
fn positive(x: int) int = {
	assert(x > 0);
	return x;
}
def BAD: int = @comptime(positive(-1)); // error: assert failed during @comptime
export fn main() void = void;
lora Allowed 
fn positive(x: int) int = {
	assert(x > 0);
	return x;
}
def OK: int = @comptime(positive(4)); // ok: assertion passes at compile time
export fn main() void = assert(OK == 4);

Recursive macro expansion or wrong macro arity

Self-recursive macros are rejected after the 64-level expansion cap, and macro invocations must match parameter count.

Compile error: Macro expansion limit exceeded / wrong number of macro arguments
lora Rejected 
macro loop_m(a) = @loop_m(a);
export fn main() void = { @loop_m(1); } // error: macro recursion limit
lora Allowed 
macro id(a) = a; // ok: non-recursive macro
export fn main() void = assert(@id(42) == 42);

Constraints & Limits

Macro Expansion Limit
64 levels
